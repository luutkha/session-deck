- export type DragItem = { readonly kind: 'session' | 'group'; readonly id: Id }
- export type DropOver = { readonly kind: 'session' | 'group' | 'ungrouped'; readonly id?: Id; readonly position: 'before' | 'after' | 'inside' }
- export type DropAction = { readonly type: 'moveSession'; readonly sessionId: Id; readonly groupId: Id | null; readonly index: number } | { readonly type: 'reorderGroups'; readonly order: readonly Id[] }
- computeDrop(drag: DragItem, over: DropOver, data: StoreData): DropAction | null
  Rules: session dropped on a group header with position 'inside' -> move to END of that group. Session dropped on another session 'before'/'after' -> that session's group, index = target's index (+1 for 'after') measured in the group's sorted list; when the dragged session is currently in the SAME group and its old index is lower than the computed index, subtract 1 (because it is removed first). Session dropped on 'ungrouped' (inside) -> end of the ungrouped list. Group dropped on a group 'before'/'after' -> reorderGroups with the full new order. Group dropped on anything else, session dropped on itself, no-op moves (result identical to current position), unknown ids -> return null.
